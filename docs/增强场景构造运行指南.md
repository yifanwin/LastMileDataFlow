# 增强场景构造运行指南

只给命令。工作目录固定为增强 worktree 根。

```bash
cd /data0/wenyifan/MoMaTrajGen/LastMileDataFlow/.worktrees/procedural-obstacle-augmentation
```

## 环境

```bash
export DATAFLOW_PYTHON="$PWD/../no-edit-lastmile-val/.venv-curobo-v080/bin/python"
export MUJOCO_GL=egl
export MPLCONFIGDIR="$PWD/outputs/dependencies/mpl"
export PYTHONUNBUFFERED=1
mkdir -p outputs/logs
```

## 1. 构造（Case1 / Case2）

```bash
# Case1：随机 1–3 个，恰 1 把地面 Chair
RUN="procedural-obstacles-case1-$(date +%Y%m%d-%H%M%S)"
MUJOCO_EGL_DEVICE_ID=<空闲卡> bin/lastmile-dataflow augment-obstacles \
  --config configs/case_edits/procedural_obstacles_val103_cup30_case1.json \
  --output-dir outputs --run-id "$RUN" \
  2>&1 | tee "outputs/logs/$RUN.log"

# Case2：随机 6–12 个，全部放目标支撑面，至少两个尺寸达标的大/高物体
RUN="procedural-obstacles-case2-$(date +%Y%m%d-%H%M%S)"
MUJOCO_EGL_DEVICE_ID=<空闲卡> bin/lastmile-dataflow augment-obstacles \
  --config configs/case_edits/procedural_obstacles_val103_cup30_case2.json \
  --output-dir outputs --run-id "$RUN" \
  2>&1 | tee "outputs/logs/$RUN.log"
```

两份配置均 `variant_count=5`。产物在 `outputs/diagnostics/<run>/`。

## 2. 审计（只读复核）

```bash
MUJOCO_GL=egl PYTHONPATH=src "$DATAFLOW_PYTHON" scripts/audit_procedural_obstacles.py \
  outputs/diagnostics/<run> --output outputs/diagnostics/<run>/audit.json
```

## 3. 重渲媒体（仅相机覆盖变更时）

```bash
MUJOCO_GL=egl PYTHONPATH=src "$DATAFLOW_PYTHON" scripts/rerender_obstacle_cases.py \
  outputs/diagnostics/<物理父run> outputs/diagnostics/<新run>
```

## 4. 构造后接入（可选，走采集用）

```bash
# cuRobo：直接用已固化的最终站位配置起采集（见 运行手册 §一.B1）
bin/lastmile-dataflow collect-no-edit \
  --assets-dir /data0/wenyifan/MoMaTrajGen/molmospaces_data/assets \
  --config configs/case_edits/no_edit_curobo_case1_v000_final_stations.json \
  --run-id "no-edit-final20-case1-v000-$(date +%Y%m%d-%H%M%S)" \
  --houses 103 --targets Cup_30 --gpu-ids <空闲卡> --workers 1

# MolmoBot：先 native 重建，再重验全部候选点，最后用 final_stations 配置采集（见 运行手册 §二.B）
PYTHONPATH=src "$DATAFLOW_PYTHON" scripts/rebuild_augmented_native.py \
  --variant outputs/diagnostics/<原增强run>/variants/vNNN --run-id <native准备run>

MUJOCO_EGL_DEVICE_ID=<空闲卡> "$DATAFLOW_PYTHON" scripts/revalidate_native_stations.py \
  --variant outputs/diagnostics/<原增强run>/variants/vNNN \
  --native-scene outputs/no_edit/<native准备run>/scenes/val_103/scene \
  --config configs/no_edit/molmobot.json \
  --output outputs/diagnostics/<唯一native重验run>
```
